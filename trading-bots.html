# Max drawdown utility
import numpy as np
import pandas as pd

def max_drawdown(equity: pd.Series) -> float:
    roll_max = equity.cummax()
    dd = equity / roll_max - 1.0
    return float(dd.min())

# Vectorized backtest stub
def backtest(df: pd.DataFrame, position: pd.Series, cost_bps: float = 1.0) -> pd.DataFrame:
    ret = df['RetSimple'].fillna(0.0)
    shift_pos = position.shift().fillna(0.0)
    gross = shift_pos * ret
    cost = (position.diff().abs().fillna(0.0)) * (cost_bps / 10000.0)
    net = gross - cost
    equity = (1.0 + net).cumprod()
    return pd.DataFrame({'ret_net': net, 'equity': equity})

Metodologia i rzetelność testów Methodology and test rigor

  • Walk‑forward — okna train/test dla walidacji czasowej; unikanie data leakagetrain/test rolling windows for time validation; avoid leakage
  • Koszty i slippage — model bps, wpływ na equity i metrykibps model impacting equity and metrics
  • Permutation test — p‑value względem reshuffle w blokachp‑value vs. block‑reshuffled baseline
